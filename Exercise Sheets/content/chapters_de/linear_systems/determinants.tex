\begin{ExerciseBlock}
\ExerciseHeading{04}{Determinanten berechnen}{linear_systems_determinants}
Berechnen Sie die Determinanten der folgenden Matrizen:
\begin{enumerate}[label=\alph*)]
	\item
	\begin{equation}
	 \begin{pmatrix}
	   1 & 2 \\
	   4 & 8 
	 \end{pmatrix}
	\end{equation}

	\item
	\begin{equation}
	 \begin{pmatrix}
	   1 & 2 & 2  \\
	   3 & -3 & 1  \\
	   1 & 4 & -4 
	 \end{pmatrix}
	\end{equation}

	\item 
	\begin{equation}
	 \begin{pmatrix}
	   1 & 0 &  0 & 0 &  0 \\
	   2 & 2 &  3 & 1 & -1 \\
	   7 & 0 &  0 & 2 &  0 \\
	   3 & 0 & -1 & 3 &  0 \\
	   4 & 2 &  5 &  2 & 0 
	 \end{pmatrix}
	\end{equation}

	\item 
	\begin{equation}
	 \begin{pmatrix}
	   d_{1} & ? & ? & ? & \cdots & ? \\
	   0 & d_{2} & ? & ? & \cdots & ? \\
	   0 & 0 & d_{3} & ? & \cdots & ? \\
	   0 & 0 & 0 & d_{4} & \cdots & ? \\
	   \vdots  & \vdots & \vdots & \vdots  & \ddots & \vdots  \\
	   0 & 0 & 0 & 0 & \cdots & d_{n} 
	 \end{pmatrix}
	\end{equation}

\end{enumerate}

\end{ExerciseBlock}

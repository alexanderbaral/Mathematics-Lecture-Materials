\ExerciseHeading{06}{Darstellung von Vektoren mit Einheitsvektoren}{vector_algebra_unit_vector_representation}
Gegeben ist der Vektor \[\vec{a}=\left(\begin{array}{c} 3 \\ 2 \\ -4 \end{array}\right).\]
\begin{enumerate}[label=\alph*)]
	\item Geben Sie den Vektor als Summe der Einheitsvektoren 
	\begin{center}
	$\vec{e}_x = \left(\begin{array}{c} 1\\0\\0 \end{array}\right)$, 
	$\vec{e}_y = \left(\begin{array}{c} 0\\1\\0 \end{array}\right)$ und
	$\vec{e}_z = \left(\begin{array}{c} 0\\0\\1 \end{array}\right)$.
	\end{center}
	an.
	\item Geben Sie den Vektor als Summe der Einheitsvektoren 
	\begin{center}
	$\vec{e}_1 = \left(\begin{array}{c} 1\\0\\0 \end{array}\right)$, 
	$\vec{e}_2 = \frac{1}{\sqrt{2}} \left(\begin{array}{c} 1\\1\\0 \end{array}\right)$ und $\vec{e}_3 = \frac{1}{\sqrt{3}} \left(\begin{array}{c} 1\\1\\1 \end{array}\right)$.
	\end{center}
	an.
\end{enumerate} 

\end{ExerciseBlock}

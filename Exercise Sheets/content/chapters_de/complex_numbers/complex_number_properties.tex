\begin{ExerciseBlock}
\ExerciseHeading{2}{Eigenschaften einer komplexen Zahl}{complex_numbers_complex_number_properties}
Bestimmen Sie für die folgenden komplexen Zahlen deren Realteil $\Re(z)$, deren Imaginärteil $\Im(z)$, deren Betrag $|z|$ und Winkel $\varphi = \arg(z)$ (Argument von $z$).
\begin{enumerate}[label=\alph*)]
	\item $z = 3 - 4i$
	
	\item $z = \frac{1+\sqrt{3}i}{\sqrt{3}+i}$
	\item $z = 4 \, e^{-i \frac{\pi}{4}}$
	\item $z = e^{2 + i \frac{\pi}{4}}$
	\item $z = \frac{ \sqrt{2} \; e^{i \frac{3}{2} \pi}}{1-i}$
	
\end{enumerate}

\end{ExerciseBlock}

\begin{enumerate}[label=\alph*)]
	\item $\int_1^2 \frac{1}{x} \, dx$
	\item $\int_0^{\pi} \sin(x) \, dx$
	\item $\int_{-\pi}^{\pi} \sin(x) \, dx$
	\item $\int_0^4 |2x-3| \, dx$
\end{enumerate}

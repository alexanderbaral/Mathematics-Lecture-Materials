\begin{ExerciseBlock}
\ExerciseHeading{1}{unbestimmte Integrale}{integral_calculus_antiderivatives}
Bestimmen Sie sämtliche Stammfunktionen. Kontrollieren Sie das Ergebnis durch Ableiten.
\begin{enumerate}[label=\alph*)]
	\item $f(x) = x^n \ \ $ mit $n \neq -1$ 
	\item $f(x) = 3x^5-2x^3+x^2+1$
	\item $f(x) = \ln|x|$
	\item $f(x) = 1$
	\item $f(x) = \frac{1}{\sqrt{x}}$
	\item $f(x) = e^{a x}$
	\item $f(x) = \cos(x) + \sin(x)$
	\item $f(x) = \frac{5}{1+x^2}$
	\item $f(x) = 2 \sqrt{x}$  
	\item $f(x) = u'(x) \; v(x) + v'(x) \; u(x)$
	\item $f(x) = \cos(x) x + \sin(x)$
\end{enumerate}

\end{ExerciseBlock}

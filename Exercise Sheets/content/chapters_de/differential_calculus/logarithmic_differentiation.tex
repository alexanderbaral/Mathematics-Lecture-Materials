\begin{ExerciseBlock}
\ExerciseHeading{08}{Ableitungsregel}{differential_calculus_logarithmic_differentiation}
Es seien $u$ und $v$ differenzierbar und $u(x)>0$. Zeigen Sie, dass die Ableitung von Funktionen der Form $f(x) = u(x)^{v(x)}$ durch  
\begin{equation*}
	f'(x) = \left( \frac{v(x)}{u(x)} u'(x) + \ln(u(x)) v'(x) \right) u(x)^{v(x)}
\end{equation*}
gegeben ist und wenden Sie dies auf die folgenden Funktionen an:
\begin{enumerate}[label=\alph*)]
	\item $f(x) = x^x$
	\item $f(x) = x^{-\ln(x)}$
\end{enumerate}

\end{ExerciseBlock}

\begin{tikzpicture}[>=Latex,line cap=round]
  \foreach \panel/\shift in {1/0,2/6,3/12} {
    \begin{scope}[xshift=\shift cm,x=0.45cm,y=1.1cm]
      \ifnum\panel=2
        \fill[black!8] (0,-1.1) rectangle (3.1416,1.1);
      \else
        \fill[black!8] (-1.5708,-1.1) rectangle (1.5708,1.1);
      \fi
      \draw[->,black!55] (-3.3,0)--(6.7,0) node[right] {$x$};
      \draw[->,black!55] (0,-1.3)--(0,1.5) node[above] {$y$};
      \foreach \xx/\lab in {-3.1416/{-\pi},3.1416/{\pi},6.2832/{2\pi}}
        \node at (\xx,-1.65) {$\lab$};
      \ifnum\panel=1
        \draw[THMBlue,line width=1.1pt,domain=-3.14:6.28,samples=120] plot (\x,{sin(\x r)});
        \node[THMBlue] at (1.5,2.25) {$\sin x$};
        \node at (1.5,-2.4) {$[-\pi/2,\pi/2]$};
      \else\ifnum\panel=2
        \draw[THMBlue,line width=1.1pt,domain=-3.14:6.28,samples=120] plot (\x,{cos(\x r)});
        \node[THMBlue] at (1.5,2.25) {$\cos x$};
        \node at (1.5,-2.4) {$[0,\pi]$};
      \else
        \begin{scope}
          \clip (-3.14,-1.1) rectangle (6.28,1.1);
          \foreach \shiftx in {-3.1416,0,3.1416,6.2832}
            \draw[THMBlue,line width=1.1pt,domain=-1.4:1.4,samples=80]
              plot ({\x+\shiftx},{tan(\x r)});
        \end{scope}
        \node[THMBlue] at (1.5,2.25) {$\tan x$};
        \node at (1.5,-2.4) {$(-\pi/2,\pi/2)$};
      \fi\fi
    \end{scope}
  }
\end{tikzpicture}

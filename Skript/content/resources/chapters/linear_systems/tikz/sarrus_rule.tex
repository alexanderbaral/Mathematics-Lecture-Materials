  entry/.style={inner sep=2pt,fill=white},

    \foreach \col in {1,2}{
      \node[entry,text=green!45!black] (r\row\col)
        at ({(\col+2.35)*\dx},{-(\row-1)*\dy}) {$a_{\row\col}$};

  \begin{pgfonlayer}{background}

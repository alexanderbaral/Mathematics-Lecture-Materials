\begin{tikzpicture}[>=Latex]
  \path[use as bounding box] (-0.15,-1.25) rectangle (15.85,3.15);

  \node[
    draw=THMBlue!70!black,
    fill=THMBlue!4,
    rounded corners=2pt,
    minimum width=6.05cm,
    minimum height=2.25cm,
    align=center
  ] (parameter) at (3.25,1.0) {%
    \Lang{Parameterform}{parametric form}\\[0.55em]
    $\displaystyle
      \vec x=\vec p+\lambda\vec a+\mu\vec b$
  };

  \node[
    draw=black!55,
    fill=black!3,
    rounded corners=2pt,
    minimum width=6.05cm,
    minimum height=2.25cm,
    align=center
  ] (normal) at (12.45,1.0) {%
    \Lang{Normalenform}{normal equation}\\[0.55em]
    $\displaystyle
      (\vec x-\vec p)\cdot\vec n=0$
  };

  \draw[->,THMBlue,line width=1.15pt]
    (parameter.north east) to[bend left=16]
    node[above=5pt,fill=white,inner sep=1.5pt]
      {$\vec n=\vec a\times\vec b$}
    (normal.north west);

  \draw[->,black!65,line width=1.0pt]
    (normal.south west) to[bend left=16]
    node[below=5pt,fill=white,inner sep=1.5pt]
      {$\vec n\cdot\vec a=\vec n\cdot\vec b=0$}
    (parameter.south east);
\end{tikzpicture}

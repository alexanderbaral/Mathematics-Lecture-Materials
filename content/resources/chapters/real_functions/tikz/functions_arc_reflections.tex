\begin{tikzpicture}[>=Latex,line cap=round]
  \foreach \panel/\shift in {1/0,2/5.85,3/11.7} {
    \begin{scope}[xshift=\shift cm,x=0.85cm,y=0.85cm]
      \draw[black!8,step=1] (-1.7,-1.7) grid (3.3,3.3);
      \draw[->,black!55] (-1.9,0)--(3.6,0) node[right] {$x$};
      \draw[->,black!55] (0,-1.9)--(0,3.6) node[above] {$y$};
      \foreach \v in {-1,1} {
        \node[anchor=north] at (\v,-2.2) {$\v$};
        \node[anchor=east] at (-2.1,\v) {$\v$};
      }
      \draw[black!45,dashed] (-1.7,-1.7)--(3.3,3.3);
      \ifnum\panel=1
        \draw[THMBlue,line width=1.1pt,domain=-1.5708:1.5708,samples=100]
          plot (\x,{sin(\x r)});
        \draw[green!55!black,line width=1.1pt,domain=-1.5708:1.5708,samples=100]
          plot ({sin(\x r)},\x);
        \node[THMBlue] at (0.7,4.8) {$\sin x$};
        \node[green!55!black] at (0.7,-3.2) {$\arcsin x$};
      \else\ifnum\panel=2
        \draw[THMBlue,line width=1.1pt,domain=0:3.1416,samples=100]
          plot (\x,{cos(\x r)});
        \draw[green!55!black,line width=1.1pt,domain=0:3.1416,samples=100]
          plot ({cos(\x r)},\x);
        \node[THMBlue] at (0.7,4.8) {$\cos x$};
        \node[green!55!black] at (0.7,-3.2) {$\arccos x$};
      \else
        \begin{scope}
          \clip (-1.7,-1.7) rectangle (3.3,3.3);
          \draw[THMBlue,line width=1.1pt,domain=-1.4:1.4,samples=100]
            plot (\x,{tan(\x r)});
          \draw[green!55!black,line width=1.1pt,domain=-1.7:3.3,samples=100]
            plot (\x,{atan(\x)*pi/180});
        \end{scope}
        \node[THMBlue] at (0.7,4.8) {$\tan x$};
        \node[green!55!black] at (0.7,-3.2) {$\arctan x$};
      \fi\fi
    \end{scope}
  }
\end{tikzpicture}

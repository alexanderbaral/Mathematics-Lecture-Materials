    \begin{scope}[xshift=\shift cm,x=1.1cm,y=0.52cm]
      \draw[black!8,step=1] (-2,-4) grid (2,4);
      \draw[->,black!55] (-2.4,0)--(2.6,0) node[right] {$x$};

      \foreach \xx in {-2,-1,1,2} {
        \draw[black!50] (\xx,-0.12)--(\xx,0.12);
        \ifnum\panel=3
          \ifnum\xx<0
            \node[above=4pt] at (\xx,0) {$\xx$};
          \else
            \node[below=4pt] at (\xx,0) {$\xx$};
          \fi
        \else
          \node[below=4pt] at (\xx,0) {$\xx$};
        \fi
      }
      \foreach \yy in {-4,-2,2,4} {
        \draw[black!50] (-0.06,\yy)--(0.06,\yy);
        \node[left=4pt] at (0,\yy) {$\yy$};
      }
      \begin{scope}
        \clip (-2,-4) rectangle (2,4);
        \draw[THMBlue,line width=1.1pt,domain=-2:2,samples=100]
          plot (\x,{pow(\x,\panel)});
      \end{scope}
      \node[THMBlue] at (0,6) {$f(x)=\title$};
    \end{scope}
  }
